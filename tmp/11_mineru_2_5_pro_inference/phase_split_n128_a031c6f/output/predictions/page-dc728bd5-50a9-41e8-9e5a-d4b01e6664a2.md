We claim that

\[
\int_ {Q} | g \lambda | \prod_ {i \in \tau} \left(\sum_ {P \in \tilde {\mathcal {S}}, P \subseteq Q} \eta_ {i, P} \chi_ {P} (x)\right) ^ {k _ {i} - t _ {i}} \lesssim \int_ {Q} A _ {\tilde {\mathcal {S}}, \eta_ {\tau (1)}} ^ {k _ {\tau (1)} - t _ {\tau (1)}} \left(A _ {\tilde {\mathcal {S}}, \eta_ {\tau (2)}} ^ {k _ {\tau (2)} - t _ {\tau (2)}} \left(A _ {\tilde {\mathcal {S}}, \eta_ {\tau (3)}} ^ {k _ {\tau (3)} - t _ {\tau (3)}} | g \lambda |\right)\right). \tag {4.27}
\]

Clearly, this follows directly from the definitions that

\[
\begin{array}{l} \int_ {Q} | g \lambda | \prod_ {i \in \tau} \Bigl (\sum_ {P \in \tilde {\mathcal {S}}, P \subseteq Q} \eta_ {i, P} \chi_ {P} (x) \Bigr) ^ {k _ {i} - t _ {i}} \\ \lesssim \int_ {Q} A _ {\tilde {\mathcal {S}}, \eta_ {\tau (1)}} ^ {k _ {\tau (1)} - Combining above estimate (4.27), we obtain

\[
\begin{array}{l} \sum_ {Q \in \mathcal {S}} \mu (Q) ^ {\eta} \left(\int_ {Q} | g \lambda | \prod_ {i \in \tau} \left| b _ {i} - b _ {i, Q} \right| ^ {k _ {i} - t _ {i}}\right) \prod_ {i \in \tau} \langle \left| f _ {i} \left(b _ {i} - b _ {i, Q}\right) ^ {t _ {i}} \right| \rangle_ {Q} \prod_ {j \in \tau} \langle | f _ {j} | \rangle_ {Q} \\ \lesssim \prod_ {i \in \tau} \| b _ {i} \| _ {\mathrm{BMO} _ {\eta_ {i}}} ^ {k _ {i}} \int_ {Q} A _ {\tilde {\mathcal {S}}, \eta_ {\tau (1)}} ^ {k _ {\tau (1)} - t _ {\tau (1)}} \big (A _ {\tilde {\mathcal {S}}, \eta_ {\tau (2)}} ^ {k _ {\tau (2)} - t _ {\tau (2)}} \big (A _ {\tilde {\mathcal {S}}, \eta_ {\tau (3)}} ^ {k _ {\tau (3)} - t _ {\tau (3)}} | g \lambda | \big) \big) \mathcal {A} _ {\eta , \tilde {\mathcal {S}}} (\vec {\mathbf {A}}) \\ \end{array}
